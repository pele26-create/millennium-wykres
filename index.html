async function generateFullData(scenario) { const liveHistory = await fetchLiveHistory(); const historyMap = {}; liveHistory.forEach(d => historyMap[d.date] = d.nav); let lastKnownNav = liveHistory.length > 0 ? liveHistory[0].nav : 92.00; if (liveHistory.length > 0) { cutoffDate = liveHistory[liveHistory.length - 1].date; } const data = []; let date = new Date(2018, 0, 1); const endDate = new Date(2028, 11, 31); let cpiIndex = 100.0; let fundAnnualReturn = 0.058; let projAnnualCpi = 0.032; if (scenario === 'optimistic') { fundAnnualReturn = 0.085; projAnnualCpi = 0.028; } else if (scenario === 'pessimistic') { fundAnnualReturn = 0.025; projAnnualCpi = 0.045; } const projDailyDrift = Math.pow(1 + fundAnnualReturn, 1/365) - 1; const projDailyCpi = Math.pow(1 + projAnnualCpi, 1/365) - 1; while (date <= endDate) { const dateStr = date.toISOString().split('T')[0]; const year = date.getFullYear(); const isProjection = dateStr > cutoffDate; if (!isProjection) { // Jeśli jest wycena w danym dniu – bierzemy ją. Jeśli nie – utrzymujemy ostatnią znaną (interpolacja dzienna) if (historyMap[dateStr]) { lastKnownNav = historyMap[dateStr]; } const annualCpi = cpiYoYMap[year] || 4.0; const dailyCpiRate = Math.pow(1 + annualCpi / 100, 1 / 365) - 1; cpiIndex = cpiIndex * (1 + dailyCpiRate); } else { lastKnownNav = lastKnownNav * (1 + projDailyDrift); cpiIndex = cpiIndex * (1 + projDailyCpi); } const dayOfWeek = date.getDay(); // Zapisujemy dni robocze (pon-pt) dla codziennego wykresu giełdowego if (dayOfWeek !== 0 && dayOfWeek !== 6) { data.push({ date: dateStr, nav: parseFloat(lastKnownNav.toFixed(2)), cpiIndex: parseFloat(cpiIndex.toFixed(3)), isProjection: isProjection }); } date.setDate(date.getDate() + 1); // Przeskok codzienny! } return data; }
